CREATE OR REPLACE FUNCTION get_or_create_id(
    table_name text,
    name_column text,
    name_value text
) RETURNS integer AS $$
DECLARE
    result_id integer;
BEGIN
    EXECUTE format('SELECT id FROM %I WHERE %I = $1', table_name, name_column) INTO result_id USING name_value;
    IF result_id IS NULL THEN
        EXECUTE format('INSERT INTO %I (%I) VALUES ($1) RETURNING id', table_name, name_column) INTO result_id USING name_value;
    END IF;
    RETURN result_id;
END;
$$ LANGUAGE plpgsql;

    SELECT
        (SELECT id FROM lesson WHERE name = 'Информатика') AS lesson_id,
        (SELECT id FROM classroom WHERE name = '8 класс') AS classroom_id,
        (SELECT id FROM lesson_stage WHERE name = 'Самостоятельная работа с самопроверкой') AS lesson_stage_id,
        (SELECT id FROM lesson_format WHERE name = 'Очный') AS lesson_format_id,
        (SELECT id FROM tech_equip WHERE name = 'ПК+Интернет') AS tech_equip_id,
        45 AS duration
)
INSERT INTO context (lesson_id, classroom_id, lesson_stage_id, lesson_format_id, tech_equip_id, duration)
SELECT lesson_id, classroom_id, lesson_stage_id, lesson_format_id, tech_equip_id, duration
FROM ctx
WHERE NOT EXISTS (
    SELECT 1 FROM context c
    WHERE c.lesson_id = ctx.lesson_id
      AND c.classroom_id = ctx.classroom_id
      AND c.lesson_stage_id = ctx.lesson_stage_id
      AND c.lesson_format_id = ctx.lesson_format_id
      AND (c.tech_equip_id = ctx.tech_equip_id OR (c.tech_equip_id IS NULL AND ctx.tech_equip_id IS NULL))
      AND c.duration = ctx.duration
);

CREATE TEMP TABLE tmp_solutions AS
SELECT DISTINCT
    sol_type, sol_strategy, sol_risk, sol_autonomy,
    (SELECT id FROM solution_type WHERE name = sol_type) AS sol_type_id,
    (SELECT id FROM response_strategy WHERE name = sol_strategy) AS resp_strat_id
FROM tmp_choices;

INSERT INTO solution (sol_type_id, resp_strat_id, risk, student_autonomy)
SELECT sol_type_id, resp_strat_id, sol_risk, sol_autonomy
FROM tmp_solutions
ON CONFLICT (sol_type_id, resp_strat_id, risk, student_autonomy) DO NOTHING;

CREATE TEMP TABLE solution_map AS
SELECT
    s.sol_type_id,
    s.resp_strat_id,
    s.risk,
    s.student_autonomy,
    s.id AS solution_id
FROM solution s
JOIN tmp_solutions ts ON ts.sol_type_id = s.sol_type_id
                     AND ts.resp_strat_id = s.resp_strat_id
                     AND ts.sol_risk = s.risk
                     AND ts.sol_autonomy = s.student_autonomy;

INSERT INTO solution_resource (resource_id, solution_id)
SELECT DISTINCT r.id, sm.solution_id
FROM tmp_choices tc
CROSS JOIN UNNEST(tc.resources) AS res_name
JOIN resource r ON r.name = res_name
JOIN solution_map sm ON sm.sol_type_id = (SELECT id FROM solution_type WHERE name = tc.sol_type)
                    AND sm.resp_strat_id = (SELECT id FROM response_strategy WHERE name = tc.sol_strategy)
                    AND sm.risk = tc.sol_risk
                    AND sm.student_autonomy = tc.sol_autonomy
ON CONFLICT (resource_id, solution_id) DO NOTHING;

    od.description,
    tc2.id
FROM tmp_choices tc
JOIN node_ids n_from ON n_from.node_id = tc.from_node
JOIN node_ids n_to ON n_to.node_id = tc.next_node
JOIN solution_map sm ON sm.sol_type_id = (SELECT id FROM solution_type WHERE name = tc.sol_type)
                     AND sm.resp_strat_id = (SELECT id FROM response_strategy WHERE name = tc.sol_strategy)
                     AND sm.risk = tc.sol_risk
                     AND sm.student_autonomy = tc.sol_autonomy
LEFT JOIN outcome_data od ON od.node_id = tc.next_node
LEFT JOIN type_consequence tc2 ON tc2.name = CASE
    WHEN od.type_name = 'failure' THEN 'Неудача'
    WHEN od.type_name = 'success' THEN 'Успех'
    WHEN od.type_name = 'partial' THEN 'ЧастичныйУспех'
END
WHERE NOT EXISTS (
    SELECT 1 FROM choice c
    WHERE c.solution_id = sm.solution_id
      AND c.from_script_node_id = n_from.id
      AND c.next_script_node_id = n_to.id
      AND c.description = tc.description
);

-- Метрики и последствия (consequence)
-- --------------------------------------------------------------------
CREATE TEMP TABLE outcome_metrics AS
SELECT * FROM (VALUES
('out_1', -0.5, 0.6, -0.4, -0.4, -0.1, 0.5),
    ('out_2', -0.3, 0.3, -0.3, -0.2, -0.1, 0.3),
    ('out_3', 0.4, -0.5, 0.5, 0.3, 0.3, -0.2),
    ('out_4', 0.2, -0.3, 0.3, 0.2, 0.1, -0.1),
    ('out_5', 0.3, -0.4, 0.4, 0.2, 0.2, -0.1),
    ('out_6', 0.3, -0.3, 0.4, 0.4, 0.3, -0.1),
    ('out_7', 0.1, -0.2, 0.2, 0.1, 0.1, 0)
) AS t(node_id, motivation, stress, trust, class_climate, teacher_authority, teacher_burnout);

INSERT INTO change_metrics (motivation, stress, trust, class_climate, teacher_authority, teacher_burnout)
SELECT motivation, stress, trust, class_climate, teacher_authority, teacher_burnout
FROM outcome_metrics
ON CONFLICT (motivation, stress, trust, class_climate, teacher_authority, teacher_burnout) DO NOTHING;

WITH inserted_metrics AS (
    SELECT om.node_id, cm.id AS change_metrics_id
    FROM outcome_metrics om
    JOIN change_metrics cm ON cm.motivation = om.motivation
                          AND cm.stress = om.stress
                          AND cm.trust = om.trust
                          AND cm.class_climate = om.class_climate
                          AND cm.teacher_authority = om.teacher_authority
                          AND cm.teacher_burnout = om.teacher_burnout
)
INSERT INTO consequence (change_metrics_id, type_conseq_id, text, long_term, reversible, intensity)
SELECT
    im.change_metrics_id,
    tc.id,
    od.description,
    od.long_term,
    od.reversible,
    0.5
FROM inserted_metrics im
JOIN outcome_data od ON od.node_id = im.node_id
JOIN type_consequence tc ON tc.name = CASE od.type_name
    WHEN 'failure' THEN 'Неудача'
    WHEN 'success' THEN 'Успех'
    WHEN 'partial' THEN 'ЧастичныйУспех'
END
ON CONFLICT (change_metrics_id, type_conseq_id, text) DO NOTHING;
