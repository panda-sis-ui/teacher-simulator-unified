BEGIN;

-- Situation
INSERT INTO situation (context_id, problem_id, script_node_id, name)
SELECT
    ctx.id,
    p.id,
    sn.id,
    'Обвинение ученика в предвзятости оценивания'
FROM context ctx
CROSS JOIN problem p
CROSS JOIN script_node sn
WHERE ctx.lesson_id = (SELECT id FROM lesson WHERE name = 'Информатика' LIMIT 1)
  AND ctx.classroom_id = (SELECT id FROM classroom WHERE name = '9 класс' LIMIT 1)
  AND p.description = 'Ученик публично заявляет о предвзятости учителя после получения низкой оценки за контрольную работу.'
  AND sn.id = 1
ON CONFLICT DO NOTHING;

-- Situation_participants
INSERT INTO situation_participants (situation_id, participant_id)
SELECT s.id, p.id
FROM situation s
CROSS JOIN participant p
WHERE s.name = 'Обвинение ученика в предвзятости оценивания'
  AND p.full_name IN ('Мария Петровна', 'Дмитрий Волков')
ON CONFLICT (situation_id, participant_id) DO NOTHING;

COMMIT;

UNION ALL SELECT 'change_metrics', COUNT(*) FROM change_metrics
UNION ALL SELECT 'script_node', COUNT(*) FROM script_node
UNION ALL SELECT 'situation', COUNT(*) FROM situation
UNION ALL SELECT 'situation_participants', COUNT(*) FROM situation_participants;
